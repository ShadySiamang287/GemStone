cmake_minimum_required(VERSION 3.28)
project(Engine LANGUAGES CXX)

include(cmake/Dependencies.cmake)
include(cmake/dxc.cmake)

add_library(Engine SHARED)

target_compile_definitions(Engine
PUBLIC 
    GEM_PLATFORM_WINDOWS
PRIVATE
    GEM_BUILD_DLL
    UNICODE
    _UNICODE
)

target_precompile_headers(Engine PRIVATE ${Engine_SOURCE_DIR}/include/pch.h)

target_sources(Engine PRIVATE
    src/Core/Application.cpp
    src/Core/Window.cpp
    src/Graphics/GraphicsContext.cpp
    src/Graphics/GraphicsCommands.cpp
    src/Graphics/GraphicsPipeline.cpp
    src/Graphics/ResourceFactory.cpp
    src/Renderer/RenderTargetManager.cpp
    src/Renderer/Renderer.cpp
    src/Core/Input.cpp
    src/ImGui/ImGuiContext.cpp
    src/Graphics/DXR/AccelerationStructs.cpp
    src/Graphics/DXR/RaytracingPipeline.cpp
)

target_include_directories(Engine
    PUBLIC ${Engine_SOURCE_DIR}/include
)

target_compile_features(Engine 
PUBLIC 
    cxx_std_23
)

target_link_libraries(Engine
PUBLIC
#FetchContent dependencies
    DirectX-Headers
    D3D12MemoryAllocator
    glm::glm
    WinPixEventRuntime
PRIVATE
    #directX
    imgui
    d3d12
    dxgi
    dxguid
)

add_custom_target(StageAgility
COMMAND ${CMAKE_COMMAND} -E make_directory $<TARGET_FILE_DIR:Engine>
COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${AGILITY_CORE}" $<TARGET_FILE_DIR:Engine>
VERBATIM)
if(AGILITY_LAYERS)
    add_custom_command(TARGET StageAgility POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${AGILITY_LAYERS}" $<TARGET_FILE_DIR:Engine>
        VERBATIM
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:WinPixEventRuntime>
            $<TARGET_FILE_DIR:Engine>
            )
endif()
add_dependencies(Engine StageAgility)
